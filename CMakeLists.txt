cmake_minimum_required(VERSION 3.20)
project(SteamLinkEyeInput VERSION 1.3.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 with MSVC is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /EHsc)
add_library(driver_steamlinkeyeinput SHARED src/driver.cpp)
target_link_libraries(driver_steamlinkeyeinput PRIVATE ws2_32 bcrypt)
add_executable(SteamLinkEyeInputControl WIN32 src/control.cpp)
target_compile_definitions(SteamLinkEyeInputControl PRIVATE UNICODE _UNICODE)
target_link_libraries(SteamLinkEyeInputControl PRIVATE user32 gdi32 comctl32 shell32 d3d11 dxgi "${CMAKE_CURRENT_SOURCE_DIR}/src/vendor/openvr_api.lib")
add_custom_command(TARGET SteamLinkEyeInputControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/package/openvr_api.dll" "$<TARGET_FILE_DIR:SteamLinkEyeInputControl>/openvr_api.dll")
include(CTest)
if(BUILD_TESTING)
  add_executable(native_layout_tests tests/native_layout_test.cpp)
  set(STEAMLINK_DRIVER_TEST_FILE "" CACHE FILEPATH "Optional local Valve DLL for read-only structural discovery tests")
  if(EXISTS "${STEAMLINK_DRIVER_TEST_FILE}")
    add_test(NAME native_layout_real_driver COMMAND native_layout_tests "${STEAMLINK_DRIVER_TEST_FILE}")
  endif()
  add_executable(center_dot_tests tests/center_dot_test.cpp)
  add_test(NAME center_dot_and_ipc COMMAND center_dot_tests)
  add_executable(native_tests tests/test.cpp)
  target_link_libraries(native_tests PRIVATE ws2_32)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/fixture/bin/win64")
  file(WRITE "${CMAKE_BINARY_DIR}/fixture/steamlinkeyeinput.ini" "[steamlinkeyeinput]\nport=19100\n")
  add_custom_command(TARGET driver_steamlinkeyeinput POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:driver_steamlinkeyeinput>" "${CMAKE_BINARY_DIR}/fixture/bin/win64/driver_steamlinkeyeinput.dll")
  add_test(NAME native_gaze_and_provider COMMAND native_tests "${CMAKE_BINARY_DIR}/fixture/bin/win64/driver_steamlinkeyeinput.dll")
  add_test(NAME control_preserves_configuration COMMAND SteamLinkEyeInputControl --self-test)
endif()
install(TARGETS driver_steamlinkeyeinput RUNTIME DESTINATION steamlinkeyeinput/bin/win64 LIBRARY DESTINATION steamlinkeyeinput/bin/win64)
install(TARGETS SteamLinkEyeInputControl RUNTIME DESTINATION .)
install(FILES package/openvr_api.dll DESTINATION .)
install(FILES LICENSE DESTINATION .)
install(DIRECTORY package/ DESTINATION . PATTERN "*.exe" EXCLUDE PATTERN "*.dll" EXCLUDE PATTERN "install-location.txt" EXCLUDE PATTERN "status.json" EXCLUDE PATTERN "status.tmp" EXCLUDE PATTERN "*.previous" EXCLUDE PATTERN "bin" EXCLUDE PATTERN "dashboard-status.json" EXCLUDE PATTERN "dashboard-registered" EXCLUDE)
